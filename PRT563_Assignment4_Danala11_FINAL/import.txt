CREATE CONSTRAINT customer_identity IF NOT EXISTS
FOR (n:Customer) REQUIRE n.customerId IS UNIQUE;

CREATE CONSTRAINT address_identity IF NOT EXISTS
FOR (n:Address) REQUIRE n.addressId IS UNIQUE;

CREATE CONSTRAINT store_identity IF NOT EXISTS
FOR (n:Store) REQUIRE n.storeId IS UNIQUE;

CREATE CONSTRAINT driver_identity IF NOT EXISTS
FOR (n:Driver) REQUIRE n.driverId IS UNIQUE;

CREATE CONSTRAINT menucategory_identity IF NOT EXISTS
FOR (n:MenuCategory) REQUIRE n.categoryId IS UNIQUE;

CREATE CONSTRAINT menuitem_identity IF NOT EXISTS
FOR (n:MenuItem) REQUIRE n.menuItemId IS UNIQUE;

CREATE CONSTRAINT menuvariant_identity IF NOT EXISTS
FOR (n:MenuVariant) REQUIRE n.variantId IS UNIQUE;

CREATE CONSTRAINT order_identity IF NOT EXISTS
FOR (n:Order) REQUIRE n.orderId IS UNIQUE;

CREATE CONSTRAINT orderitem_identity IF NOT EXISTS
FOR (n:OrderItem) REQUIRE (n.orderId, n.lineNumber) IS UNIQUE;

CREATE CONSTRAINT payment_identity IF NOT EXISTS
FOR (n:Payment) REQUIRE n.paymentId IS UNIQUE;

CREATE CONSTRAINT delivery_identity IF NOT EXISTS
FOR (n:Delivery) REQUIRE n.deliveryId IS UNIQUE;

CREATE INDEX menuvariant_price IF NOT EXISTS
FOR (v:MenuVariant) ON (v.currentPrice);

// Customer
LOAD CSV WITH HEADERS FROM 'file:///customer.csv' AS row
MERGE (n:Customer {customerId: toInteger(row.CustomerID)})
SET n.firstName = CASE WHEN row.FirstName = '' THEN null ELSE row.FirstName END,
    n.lastName = CASE WHEN row.LastName = '' THEN null ELSE row.LastName END,
    n.email = CASE WHEN row.Email = '' THEN null ELSE row.Email END,
    n.phone = CASE WHEN row.Phone = '' THEN null ELSE row.Phone END,
